function calcularPontosFlexao(sexo, idade, rep) {
    if (sexo === 'M') {
        // Tabela Exata OIC 02 (Masculino - Anexo VI)
        let minRep = 10;
        if (idade <= 20) minRep = 25;
        else if (idade <= 30) minRep = 22;
        else if (idade <= 34) minRep = 20;
        else if (idade <= 38) minRep = 17;
        else if (idade <= 41) minRep = 15;
        else if (idade <= 43) minRep = 14;
        else if (idade <= 49) minRep = 13;
        else if (idade <= 52) minRep = 12;
        else minRep = 11;

        if (rep < minRep) return 0.0;
        // Mapeamento proporcional de pontos (0 a 10) baseado nas faixas oficiais
        let excesso = rep - minRep;
        let pontos = 2.0 + (excesso * 0.35);
        return Math.max(0.0, Math.min(10.0, pontos));
    } else {
        // Tabela Exata OIC 06 - Sem apoio dos joelhos (Feminino 2026 - Anexo VI)
        let minRepF = 11;
        if (idade <= 21) minRepF = 11;
        else if (idade <= 25) minRepF = 12;
        else if (idade <= 29) minRepF = 11;
        else if (idade <= 33) minRepF = 10;
        else if (idade <= 37) minRepF = 9;
        else if (idade <= 41) minRepF = 8;
        else if (idade <= 45) minRepF = 7;
        else if (idade <= 49) minRepF = 6;
        else minRepF = 5;

        if (rep < minRepF) return 0.0;
        let excessoF = rep - minRepF;
        let pontosF = 2.0 + (excessoF * 0.7);
        return Math.max(0.0, Math.min(10.0, pontosF));
    }
}

function calcularPontosAbdominal(sexo, idade, rep) {
    if (sexo === 'M') {
        // Tabela Exata OIC 03 (Masculino - Anexo VI)
        let minAbm = 37;
        if (idade <= 27) minAbm = 37;
        else if (idade <= 30) minAbm = 35;
        else if (idade <= 33) minAbm = 32;
        else if (idade <= 35) minAbm = 31;
        else if (idade <= 38) minAbm = 30;
        else if (idade <= 41) minAbm = 27;
        else if (idade <= 44) minAbm = 25;
        else if (idade <= 50) minAbm = 22;
        else minAbm = 20;

        if (rep < minAbm) return 0.0;
        let excesso = rep - minAbm;
        let pontos = 2.0 + (excesso * 0.30);
        return Math.max(0.0, Math.min(10.0, pontos));
    } else {
        // Tabela Exata OIC 07 (Feminino - Anexo VI)
        let minAbmF = 30;
        if (idade <= 23) minAbmF = 30;
        else if (idade <= 25) minAbmF = 29;
        else if (idade <= 28) minAbmF = 27;
        else if (idade <= 30) minAbmF = 25;
        else if (idade <= 34) minAbmF = 21;
        else if (idade <= 38) minAbmF = 20;
        else if (idade <= 41) minAbmF = 17;
        else if (idade <= 45) minAbmF = 15;
        else if (idade <= 49) minAbmF = 12;
        else minAbmF = 11;

        if (rep < minAbmF) return 0.0;
        let excessoF = rep - minAbmF;
        let pontosF = 2.0 + (excessoF * 0.32);
        return Math.max(0.0, Math.min(10.0, pontosF));
    }
}

function calcularPontosCorrida(sexo, idade, dist) {
    if (sexo === 'M') {
        // Tabela Exata OIC 04 (Masculino - Anexo VI)
        let minCor = 2130;
        if (idade <= 29) minCor = 2130;
        else if (idade <= 31) minCor = 2080;
        else if (idade <= 35) minCor = 2050;
        else if (idade <= 38) minCor = 2010;
        else if (idade <= 41) minCor = 2000;
        else if (idade <= 43) minCor = 1960;
        else if (idade <= 47) minCor = 1910;
        else minCor = 1890;

        if (dist < minCor) return 0.0;
        let excesso = dist - minCor;
        let pontos = 10.0 + (excesso / 20); // Escala proporcional otimizada para até 50 pontos
        return Math.max(0.0, Math.min(50.0, pontos));
    } else {
        // Tabela Exata OIC 08 (Feminino - Anexo VI)
        let minCorF = 1760;
        if (idade <= 29) minCorF = 1760;
        else if (idade <= 33) minCorF = 1690;
        else if (idade <= 38) minCorF = 1650;
        else if (idade <= 44) minCorF = 1600;
        else minCorF = 1540;

        if (dist < minCorF) return 0.0;
        let excessoF = dist - minCorF;
        let pontosF = 10.0 + (excessoF / 20);
        return Math.max(0.0, Math.min(50.0, pontosF));
    }
}

function calcularTACF() {
    const sexo = document.getElementById('sexo').value;
    const idade = parseInt(document.getElementById('idade').value);
    const estatura = parseFloat(document.getElementById('estatura').value);
    const cintura = parseFloat(document.getElementById('cintura').value);
    const flexao = parseInt(document.getElementById('flexao').value);
    const abdominal = parseInt(document.getElementById('abdominal').value);
    const corrida = parseInt(document.getElementById('corrida').value);

    if (isNaN(idade) || isNaN(estatura) || isNaN(cintura) || isNaN(flexao) || isNaN(abdominal) || isNaN(corrida)) {
        alert("Por favor, preencha todos os campos corretamente.");
        return;
    }

    const pCintura = calcularPontosCintura(sexo, estatura, cintura);
    const pFlexao = calcularPontosFlexao(sexo, idade, flexao);
    const pAbdominal = calcularPontosAbdominal(sexo, idade, abdominal);
    const pCorrida = calcularPontosCorrida(sexo, idade, corrida);

    const grauFinal = pCintura + pFlexao + pAbdominal + pCorrida;

    const zerouTeste = (pCintura === 0 || pFlexao === 0 || pAbdominal === 0 || pCorrida === 0);
    const apto = (grauFinal >= 20.0) && !zerouTeste;

    let conceito = "";
    if (grauFinal >= 90.0) conceito = "Excelente (E)";
    else if (grauFinal >= 70.0) conceito = "Muito Bom (MB)";
    else if (grauFinal >= 40.0) conceito = "Bom (B)";
    else if (grauFinal >= 20.0) conceito = "Satisfatório (S)";
    else conceito = "Insatisfatório (I)";

    document.getElementById('ptsCintura').innerText = pCintura.toFixed(1) + " pts";
    document.getElementById('ptsFlexao').innerText = pFlexao.toFixed(1) + " pts";
    document.getElementById('ptsAbdominal').innerText = pAbdominal.toFixed(1) + " pts";
    document.getElementById('ptsCorrida').innerText = pCorrida.toFixed(1) + " pts";
    document.getElementById('grauFinal').innerText = grauFinal.toFixed(1);
    document.getElementById('conceitoGlobal').innerText = conceito;

    const statusBadge = document.getElementById('statusBadge');
    if (apto) {
        statusBadge.innerText = "APTO";
        statusBadge.className = "status-badge status-apto";
        document.getElementById('reprovacaoAlert').style.display = "none";
    } else {
        statusBadge.innerText = "NÃO APTO";
        statusBadge.className = "status-badge status-nao-apto";
        if (zerouTeste) {
            document.getElementById('reprovacaoAlert').style.display = "block";
        } else {
            document.getElementById('reprovacaoAlert').style.display = "none";
        }
    }

    document.getElementById('resultado').style.display = "block";
}
